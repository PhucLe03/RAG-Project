export default function Layout() {
  const [activeSlot, setActiveSlot] = useState(null);

  useEffect(() => {
    if (process.browser) {
      // Check the current pathname and set the active slot accordingly
      setActiveSlot((pathname => {
        switch (pathname) {
          case '/': return <Home />;
          case '/admin': return <Admin />;
          default: return <ErrorPage statusCode={404} />;
        }

  }, [])

      {activeSlot ? activeSlot : <Loading />}
    </>
  )
}
